import pandas as pd

def analyze_index(snapshot_df, index_name="SPX"):
    """
    Analyze pin/GEX snapshot for one index.
    snapshot_df: DataFrame with columns:
      ['timestamp_utc','pin_strike','spot','distance_pct',
       'pull_strength','net_gex','call_gex','put_gex']
    """

    latest = snapshot_df.iloc[-1]
    pin = latest['pin_strike']
    spot = latest['spot']
    deviation = latest['distance_pct']
    pull = latest['pull_strength']
    call_gex = latest['call_gex']
    put_gex = latest['put_gex']

    signals = []

    # Pin migration
    if snapshot_df['pin_strike'].nunique() > 1:
        signals.append("⚠️ Pin migrated intraday — reset close expectation.")

    # Spot vs pin deviation
    if deviation > 0.02:
        signals.append("⬆️ Spot consistently above pin → upward bias.")
    elif deviation < -0.02:
        signals.append("⬇️ Spot consistently below pin → downward bias.")

    # Gamma balance
    if call_gex > 3 * put_gex:
        signals.append("📈 Call GEX dominant (>3x) → upside drift risk.")
    elif put_gex > 3 * call_gex:
        signals.append("📉 Put GEX dominant (>3x) → downside drift risk.")

    # Pull strength
    if pull > 90:
        signals.append("🧲 Strong pin magnet (>90) → expect reversion.")
    elif pull < 70:
        signals.append("⚠️ Weak pin magnet (<70) → expect drift away.")

    # Final-hour overlay
    if deviation > 0.02 and call_gex > put_gex:
        signals.append("🎯 Final-hour drift overlay → higher close probability.")
    elif deviation < -0.02 and put_gex > call_gex:
        signals.append("🎯 Final-hour drift overlay → lower close probability.")

    return {
        "index": index_name,
        "pin": pin,
        "spot": spot,
        "expected_close_range": f"{pin} ± 5 pts",
        "signals": signals
    }
