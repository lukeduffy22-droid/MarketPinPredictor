1. Historical Gamma Builder

tools/historical_gamma.py

import polygon
from datetime import datetime
from app.core.gex import compute_contract_gamma, aggregate_gamma

client = polygon.RESTClient()

def build_historical_snapshot(symbol, date):
    # 1. Get underlying close
    spot = client.get_index_close(symbol, date)

    # 2. Get option contracts for same-day expiry
    contracts = client.list_options_contracts(
        underlying_ticker=symbol,
        expiration_date=date,
        limit=1000
    )

    gamma_rows = []

    for c in contracts:
        # 3. Historical IV
        iv = client.get_option_iv(c.ticker, date)
        if iv is None:
            continue

        # 4. Open interest (proxy)
        oi = c.open_interest or 0
        if oi == 0:
            continue

        gamma = compute_contract_gamma(
            spot=spot,
            strike=c.strike_price,
            iv=iv,
            time_to_expiry=1/252,
            open_interest=oi
        )

        gamma_rows.append({
            "strike": c.strike_price,
            "gamma": gamma,
            "oi": oi
        })

    pin = aggregate_gamma(spot, gamma_rows)

    return {
        "symbol": symbol,
        "date": date,
        "spot": spot,
        "pin": pin,
        "contracts": gamma_rows,
        "historical": True
    }

def validate(snapshot, actual_close):
    error = actual_close - snapshot["pin"]

    print(f"Date: {snapshot['date']}")
    print(f"Pin: {snapshot['pin']}")
    print(f"Close: {actual_close}")
    print(f"Error: {error}")
