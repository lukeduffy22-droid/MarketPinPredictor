Latency/heap guard:

# app/utils/metrics.py
import time, tracemalloc, logging
log = logging.getLogger("perf")

def timed(name):
    def deco(fn):
        def inner(*a, **k):
            t0 = time.perf_counter()
            out = fn(*a, **k)
            dt = (time.perf_counter()-t0)*1000
            if dt > 200: log.warning("SLOW %s %.1fms", name, dt)
            return out
        return inner
    return deco

tracemalloc.start()

def snapshot(label):
    cur, peak = tracemalloc.get_traced_memory()
    log.info("HEAP %s cur=%.1fMB peak=%.1fMB", label, cur/1e6, peak/1e6)

Deterministic τ-weights and composite:

# app/models/close_nowcast.py
def tau_weights(tau_min):
    wv = min(1.0, tau_min/120.0)
    wm = min(1.0, 60.0/(tau_min+1.0))
    wg = min(1.0, 90.0/(tau_min+1.0))
    wf = min(1.0, 180.0/(tau_min+1.0))
    return wv, wm, wg, wf

def combine(p_now, s_vwap, dp_micro, pin_pull, flow_sig, betas, tau_min):
    bv, bg, bf = betas["vwap"], betas["gamma"], betas["flow"]
    wv, wm, wg, wf = tau_weights(tau_min)
    num = wv*bv*s_vwap + wm*dp_micro + wg*bg*pin_pull + wf*bf*flow_sig
    den = max(1e-9, wv+wm+wg+wf)
    delta = num/den
    return p_now + delta, delta, {"vwap_pull": bv*s_vwap, "microtrend": dp_micro,
                                  "gamma_pin": bg*pin_pull, "flow": bf*flow_sig}

def bucket_for_tau(tau):
    for hi, lo in TAU_BUCKETS:
        if lo < tau <= hi: return f"{hi}_{lo}"
    return "gt60"

def confidence(rmse_bucket, rmse_max):
    rmse_max = max(1e-9, rmse_max)
    return max(0, min(100, int(round(100*(1 - rmse_bucket/rmse_max)))))

# app/features/gamma_pin.py
def pin_scores(price, oi_calls, oi_puts):
    scores = {}
    for K in sorted(set(oi_calls)|set(oi_puts)):
        oi = oi_calls.get(K,0)+oi_puts.get(K,0)
        scores[K] = oi / (1.0 + abs(K - price))
    return scores

def signed_pull(price, scores):
    # softmax over ±2% window
    ks = [k for k in scores if abs(k-price)/price <= 0.02]
    if not ks: return 0.0
    import math
    vals = [scores[k] for k in ks]
    m = max(vals)
    exps = [math.exp(v-m) for v in vals]
    Z = sum(exps)
    ek = sum(k*e for k,e in zip(ks, exps))/Z
    return ek - price

# app/features/flow_urgency.py
def flow_urgency(last_60s_trades):
    # trades: [{is_call, notional, aggressor}] aggressor in {+1 buy, −1 sell}
    net = 0.0
    for t in last_60s_trades:
        sgn = (1 if t["is_call"] else -1) * t["aggressor"]
        net += sgn * t["notional"]
    # squash to −1..1 using rolling scale s
    s = max(1.0, rolling_p75_abs_notional())
    x = max(-1.0, min(1.0, net / (5*s)))
    return x, percentile_5day(abs(net))
