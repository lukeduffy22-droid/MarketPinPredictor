# app/core/gex.py
from __future__ import annotations

from dataclasses import dataclass
from typing import Optional


@dataclass(frozen=True)
class GexResult:
    net_gex: float
    total_gex: float
    call_gex: float
    put_gex: float


def compute_gex(
    gamma_call: float,
    oi_call: float,
    gamma_put: float,
    oi_put: float,
    multiplier: float = 100.0,
) -> GexResult:
    """
    Canonical GEX computation.

    Rules:
    - Gamma inputs must be unsigned magnitudes (>= 0).
    - Exposure is computed separately for calls and puts.
    - Sign convention applied after exposure:
        net_gex = +call_gex - put_gex
        total_gex = |call_gex| + |put_gex|
    - Invariant: total_gex >= abs(net_gex)
    """
    # Defensive checks (fail closed; do not silently proceed on bad inputs)
    for name, v in (
        ("gamma_call", gamma_call),
        ("gamma_put", gamma_put),
        ("oi_call", oi_call),
        ("oi_put", oi_put),
        ("multiplier", multiplier),
    ):
        if v is None:
            raise ValueError(f"{name} is None")
        if not isinstance(v, (int, float)):
            raise TypeError(f"{name} must be numeric, got {type(v)}")

    if gamma_call < 0 or gamma_put < 0:
        raise ValueError("Gamma inputs must be unsigned magnitudes (>= 0).")
    if oi_call < 0 or oi_put < 0:
        raise ValueError("Open interest must be >= 0.")
    if multiplier <= 0:
        raise ValueError("Multiplier must be > 0.")

    call_gex = float(gamma_call) * float(oi_call) * float(multiplier)
    put_gex = float(gamma_put) * float(oi_put) * float(multiplier)

    net_gex = call_gex - put_gex
    total_gex = abs(call_gex) + abs(put_gex)

    # Hard invariant check
    if total_gex + 1e-12 < abs(net_gex):
        raise AssertionError("Invariant violated: total_gex must be >= abs(net_gex)")

    return GexResult(net_gex=net_gex, total_gex=total_gex, call_gex=call_gex, put_gex=put_gex)


This module becomes the only source of truth for net/total GEX.

from app.core.gex import compute_gex

def test_gex_invariants_and_sign():
    # Case: equal OI and gamma => net 0, total > 0
    r = compute_gex(gamma_call=0.01, oi_call=1000, gamma_put=0.01, oi_put=1000, multiplier=100)
    assert abs(r.net_gex) < 1e-9
    assert r.total_gex > 0
    assert r.total_gex >= abs(r.net_gex)

    # Case: calls dominate => net positive
    r = compute_gex(gamma_call=0.02, oi_call=2000, gamma_put=0.01, oi_put=1000, multiplier=100)
    assert r.net_gex > 0
    assert r.total_gex >= abs(r.net_gex)

    # Case: puts dominate => net negative
    r = compute_gex(gamma_call=0.01, oi_call=1000, gamma_put=0.02, oi_put=2000, multiplier=100)
    assert r.net_gex < 0
    assert r.total_gex >= abs(r.net_gex)
