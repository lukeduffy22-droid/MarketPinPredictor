import sys
import json
import pandas as pd
import numpy as np
from typing import List

import torch
import torch.nn as nn
from torch.utils.data import Dataset, DataLoader

# -----------------------------
# 1. Config
# -----------------------------

TARGET_COLUMN = "next_close"   # what we want to predict
INDEX_FILTER = "S&P 500 (SPX)" # or "NASDAQ 100 (NDX)" – change per run
BATCH_SIZE = 64
EPOCHS = 40
LR = 1e-3
VALIDATION_SPLIT = 0.2
RANDOM_SEED = 42

torch.manual_seed(RANDOM_SEED)
np.random.seed(RANDOM_SEED)

# -----------------------------
# 2. Dataset
# -----------------------------

class TabularDataset(Dataset):
    def __init__(self, X: np.ndarray, y: np.ndarray):
        self.X = torch.tensor(X, dtype=torch.float32)
        self.y = torch.tensor(y, dtype=torch.float32).view(-1, 1)

    def __len__(self):
        return len(self.X)

    def __getitem__(self, idx):
        return self.X[idx], self.y[idx]

# -----------------------------
# 3. Model
# -----------------------------

class MLP(nn.Module):
    def __init__(self, input_dim: int):
        super().__init__()
        self.net = nn.Sequential(
            nn.Linear(input_dim, 64),
            nn.ReLU(),
            nn.Linear(64, 64),
            nn.ReLU(),
            nn.Linear(64, 1)
        )

    def forward(self, x):
        return self.net(x)

# -----------------------------
# 4. Utility Functions
# -----------------------------

def select_feature_columns(df: pd.DataFrame) -> List[str]:
    """
    Automatically select numeric feature columns, excluding targets
    and obviously non-feature columns.
    """
    exclude = {
        TARGET_COLUMN,
        "predicted_price",
        "confidence",
        "index_name",
        "timestamp",
        "prediction_date",
        "Time",  # some exports use this
    }

    numeric_cols = df.select_dtypes(include=[np.number]).columns.tolist()
    feature_cols = [c for c in numeric_cols if c not in exclude]
    return feature_cols

def train_one_model(csv_path: str, model_output_path: str, meta_output_path: str):
    # -----------------------------
    # 5. Load data
    # -----------------------------
    print(f"[INFO] Loading data from {csv_path}")
    df = pd.read_csv(csv_path)

    # Optionally filter to a specific index (SPX vs NDX)
    if "index_name" in df.columns:
        df = df[df["index_name"] == INDEX_FILTER].copy()
        print(f"[INFO] Filtered to index_name = {INDEX_FILTER}, rows = {len(df)}")

    # Drop rows without target
    df = df.dropna(subset=[TARGET_COLUMN])
    print(f"[INFO] After dropping rows with missing {TARGET_COLUMN}: {len(df)} rows")

    # -----------------------------
    # 6. Feature selection
    # -----------------------------
    feature_cols = select_feature_columns(df)
    print(f"[INFO] Using {len(feature_cols)} feature columns:")
    for c in feature_cols:
        print(f"   - {c}")

    X = df[feature_cols].values.astype(np.float32)
    y = df[TARGET_COLUMN].values.astype(np.float32)

    # -----------------------------
    # 7. Train/Validation Split
    # -----------------------------
    n = len(X)
    idx = np.arange(n)
    np.random.shuffle(idx)

    split = int(n * (1 - VALIDATION_SPLIT))
    train_idx, val_idx = idx[:split], idx[split:]

    X_train, y_train = X[train_idx], y[train_idx]
    X_val, y_val = X[val_idx], y[val_idx]

    train_ds = TabularDataset(X_train, y_train)
    val_ds = TabularDataset(X_val, y_val)

    train_loader = DataLoader(train_ds, batch_size=BATCH_SIZE, shuffle=True)
    val_loader = DataLoader(val_ds, batch_size=BATCH_SIZE, shuffle=False)

    # -----------------------------
    # 8. Model/Optimizer/Loss
    # -----------------------------
    model = MLP(input_dim=X.shape[1])
    optimizer = torch.optim.Adam(model.parameters(), lr=LR)
    criterion = nn.MSELoss()

    # -----------------------------
    # 9. Training Loop
    # -----------------------------
    print("[INFO] Starting training...")
    for epoch in range(1, EPOCHS + 1):
        model.train()
        train_losses = []

        for xb, yb in train_loader:
            optimizer.zero_grad()
            preds = model(xb)
            loss = criterion(preds, yb)
            loss.backward()
            optimizer.step()
            train_losses.append(loss.item())

        # Validation
        model.eval()
        val_losses = []
        with torch.no_grad():
            for xb, yb in val_loader:
                preds = model(xb)
                loss = criterion(preds, yb)
                val_losses.append(loss.item())

        print(
            f"Epoch {epoch:03d}/{EPOCHS} "
            f"Train Loss: {np.mean(train_losses):.4f} "
            f"Val Loss: {np.mean(val_losses):.4f}"
        )

    # -----------------------------
    # 10. Save Model + Metadata
    # -----------------------------
    torch.save(model.state_dict(), model_output_path)
    print(f"[INFO] Saved model to {model_output_path}")

    meta = {
        "feature_columns": feature_cols,
        "target_column": TARGET_COLUMN,
        "index_filter": INDEX_FILTER,
        "normalization": "none (add later if needed)",
    }
    with open(meta_output_path, "w") as f:
        json.dump(meta, f, indent=2)
    print(f"[INFO] Saved metadata to {meta_output_path}")

# -----------------------------
# 11. Entry Point
# -----------------------------

if __name__ == "__main__":
    if len(sys.argv) < 2:
        print("Usage: python train_gamma_model.py <input_csv>")
        sys.exit(1)

    input_csv = sys.argv[1]
    model_out = "gamma_model_spx.pt"
    meta_out = "gamma_model_spx_meta.json"

    train_one_model(input_csv, model_out, meta_out)
