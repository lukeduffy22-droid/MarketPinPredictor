from collections import deque
from typing import Deque, Tuple, Any

class Ring1s:
    def __init__(self, capacity:int=5400): self.q: Deque[Tuple[int, Any]] = deque(maxlen=capacity)
    def add(self, ts:int, v:Any): self.q.append((ts,v))
    def view(self): return self.q
