# app/features/microtrend.py
# Lightweight ridge nowcast using rolling 1s bars from your ring buffer.
# Predicts remaining move to close over τ minutes.
from __future__ import annotations
import math
from typing import Deque, Tuple
import numpy as np

# bars: deque[(ts, price)]
def _returns_and_vol(bars: Deque[Tuple[float, float]], win: int) -> tuple[float, float]:
    if len(bars) < win + 1:
        return 0.0, 0.0
    px = np.array([p for _, p in list(bars)[-win-1:]], dtype=float)
    r = np.diff(np.log(px))
    if r.size == 0:
        return 0.0, 0.0
    return float(r.sum()), float(r.std(ddof=1)) if r.size > 1 else 0.0

def build_features(bars: Deque[Tuple[float, float]]) -> np.ndarray:
    # windows in seconds: approx 1m, 3m, 5m, 15m given 1s bars
    feats = []
    for w in (60, 180, 300, 900):
        rr, vol = _returns_and_vol(bars, w)
        feats.extend([rr, vol])
    feats.append(1.0)  # bias
    return np.array(feats, dtype=float)  # shape (9,)

class RidgeNowcaster:
    def __init__(self, lam: float = 10.0):
        # θ ∈ R^9 for features above
        self.theta = np.zeros(9)
        self.lam = lam
        self.is_fit = False

    def fit_from_history(self, X: np.ndarray, y: np.ndarray):
        # Closed-form ridge: θ = (XᵀX + λI)⁻¹ Xᵀ y
        n_feat = X.shape[1]
        A = X.T @ X + self.lam * np.eye(n_feat)
        b = X.T @ y
        self.theta = np.linalg.solve(A, b)
        self.is_fit = True

    def predict_delta(self, x: np.ndarray) -> float:
        if not self.is_fit:
            # conservative fallback → tiny mean-reversion to zero
            return float(x @ self.theta)
        return float(x @ self.theta)

# Helper to build training pairs from your persisted per-minute snapshots
def build_training_matrix(rows: list[dict]) -> tuple[np.ndarray, np.ndarray]:
    # rows should contain features at t and realized (close - price_t)
    X = []
    y = []
    for r in rows:
        X.append([
            r.get("r_1m", 0.0), r.get("vol_1m", 0.0),
            r.get("r_3m", 0.0), r.get("vol_3m", 0.0),
            r.get("r_5m", 0.0), r.get("vol_5m", 0.0),
            r.get("r_15m", 0.0), r.get("vol_15m", 0.0),
            1.0
        ])
        y.append(r.get("close_delta", 0.0))
    return np.array(X, dtype=float), np.array(y, dtype=float)
