from datetime import datetime, time
import pytz

ET = pytz.timezone("US/Eastern")

def now_et():
    return datetime.now(ET)

def market_is_open():
    t = now_et().time()
    return time(9, 30) <= t < time(16, 0)

def market_is_closed():
    return not market_is_open()

3. OFFLINE GAMMA RECOMPUTE TOOL (CRITICAL)
Purpose

Recalculate gamma with zero APIs to prove math correctness.

Create CLI tool

tools/recompute_gamma.py

import json
import sys
from app.core.gex import compute_aggregate_gamma

def main(path):
    with open(path) as f:
        snapshot = json.load(f)

    result = compute_aggregate_gamma(
        spot=snapshot["spot"],
        contracts=snapshot["contracts"]
    )

    print("Stored pin:", snapshot["pin"])
    print("Recomputed pin:", result["pin"])
    print("Delta:", result["pin"] - snapshot["pin"])

if __name__ == "__main__":
    main(sys.argv[1])

Acceptance criteria

Recomputed pin == stored pin (± epsilon)

Any mismatch is a bug, not noise
